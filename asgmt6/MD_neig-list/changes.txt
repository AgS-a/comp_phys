void neighbour_list(double x_pos[], double y_pos[], double z_pos[], double KbT, double dt,
                double cutoff_r, double mass, int n_iter, int no_of_part, double box_size) 
{
        double v_avg = sqrt((8*KbT)/(3*mass));
        double dr = 2 * 5 * v_avg * (double)n_iter * dt;
        double cut = dr + cutoff_r;

        int **neighbour_list = (int **)malloc(no_of_part * sizeof(int *));
        for (int i = 0; i < no_of_part; i++) {
            neighbour_list[i] = (int *)malloc(no_of_part * sizeof(int));
        }

        for(int i=0; i<no_of_part; i++){
                for(int j=0; j<no_of_part; j++){
                        double distance = calc_dist(box_size, x_pos[i], x_pos[j], y_pos[i], 
                                        y_pos[j], z_pos[i], z_pos[j]);
                        if(distance <= cut){
                                neighbour_list[i][j] = 1;
                        } else{
                                neighbour_list[i][j] = 0;
                        }
                }
        }
        free(neighbour_list);
}
